-- ============================================
-- MÓDULO 1: MULTITENANT & RBAC (3NF)
-- ============================================

CREATE TABLE sl_tenants (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL UNIQUE,
    plan_type VARCHAR(20) DEFAULT 'basic',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE sl_roles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id UUID NOT NULL REFERENCES sl_tenants(id) ON DELETE CASCADE,
    name VARCHAR(50) NOT NULL,
    description TEXT,
    UNIQUE(tenant_id, name)
);

CREATE TABLE sl_permissions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    resource VARCHAR(50) NOT NULL,
    action VARCHAR(50) NOT NULL,
    description TEXT,
    UNIQUE(resource, action)
);

CREATE TABLE sl_users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id UUID NOT NULL REFERENCES sl_tenants(id) ON DELETE CASCADE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE sl_user_roles (
    user_id UUID NOT NULL REFERENCES sl_users(id) ON DELETE CASCADE,
    role_id UUID NOT NULL REFERENCES sl_roles(id) ON DELETE CASCADE,
    assigned_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (user_id, role_id)
);

CREATE TABLE sl_role_permissions (
    role_id UUID NOT NULL REFERENCES sl_roles(id) ON DELETE CASCADE,
    permission_id UUID NOT NULL REFERENCES sl_permissions(id) ON DELETE CASCADE,
    PRIMARY KEY (role_id, permission_id)
);

-- ============================================
-- MÓDULO 2: SOCIAL LISTENING & SCRAPECREATORS (3NF)
-- ============================================

CREATE TABLE sl_monitored_targets (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id UUID NOT NULL REFERENCES sl_tenants(id) ON DELETE CASCADE,
    platform VARCHAR(30) NOT NULL,
    external_profile_id VARCHAR(100) NOT NULL,
    handle VARCHAR(100) NOT NULL,
    status VARCHAR(20) DEFAULT 'active',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(tenant_id, platform, external_profile_id)
);

CREATE TABLE sl_social_posts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id UUID NOT NULL REFERENCES sl_tenants(id) ON DELETE CASCADE,
    target_id UUID NOT NULL REFERENCES sl_monitored_targets(id) ON DELETE CASCADE,
    external_post_id VARCHAR(100) NOT NULL,
    content_text TEXT,
    published_at TIMESTAMP WITH TIME ZONE,
    scraped_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(external_post_id)
);

CREATE TABLE sl_post_metrics (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    post_id UUID NOT NULL REFERENCES sl_social_posts(id) ON DELETE CASCADE,
    likes_count INTEGER DEFAULT 0,
    comments_count INTEGER DEFAULT 0,
    shares_count INTEGER DEFAULT 0,
    views_count INTEGER DEFAULT 0,
    fetched_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE sl_scrape_tasks (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id UUID NOT NULL REFERENCES sl_tenants(id) ON DELETE CASCADE,
    target_id UUID NOT NULL REFERENCES sl_monitored_targets(id) ON DELETE CASCADE,
    scrape_type VARCHAR(30) DEFAULT 'full',
    api_endpoint VARCHAR(255),
    status VARCHAR(20) DEFAULT 'active',
    last_execution TIMESTAMP WITH TIME ZONE,
    next_execution TIMESTAMP WITH TIME ZONE
);

-- ============================================
-- ÍNDICES DE RENDIMIENTO
-- ============================================

CREATE INDEX idx_users_tenant ON sl_users(tenant_id);
CREATE INDEX idx_roles_tenant ON sl_roles(tenant_id);
CREATE INDEX idx_targets_tenant ON sl_monitored_targets(tenant_id);
CREATE INDEX idx_posts_tenant ON sl_social_posts(tenant_id);
CREATE INDEX idx_posts_target ON sl_social_posts(target_id);
CREATE INDEX idx_metrics_post ON sl_post_metrics(post_id);
CREATE INDEX idx_tasks_tenant ON sl_scrape_tasks(tenant_id);
CREATE INDEX idx_tasks_target ON sl_scrape_tasks(target_id);
