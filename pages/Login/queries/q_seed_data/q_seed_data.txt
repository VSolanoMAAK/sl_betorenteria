-- Analyst: permisos de monitoreo y reportes (sin gestión de usuarios)
INSERT INTO sl_role_permissions (role_id, permission_id)
SELECT r.id, p.id 
FROM sl_roles r, sl_permissions p 
WHERE r.name = 'analyst' 
AND r.tenant_id = (SELECT id FROM sl_tenants WHERE name = 'Demo Tenant')
AND p.resource IN ('dashboard', 'targets', 'posts', 'metrics', 'reports')
AND p.action IN ('view', 'create', 'edit', 'export');

-- Viewer: solo lectura

FROM sl_roles r, sl_permissions p 
WHERE r.name = 'viewer'
