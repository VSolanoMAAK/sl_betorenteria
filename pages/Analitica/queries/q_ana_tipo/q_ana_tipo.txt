WITH latest AS (
  SELECT DISTINCT ON (post_id) post_id, likes_count, comments_count
  FROM sl_post_metrics
  ORDER BY post_id, fetched_at DESC
)
SELECT p.post_type,
       count(*)::int AS posts,
       round(avg(coalesce(l.likes_count,0)),1) AS reacciones_prom,
       round(avg(coalesce(l.comments_count,0)),1) AS comentarios_prom
FROM sl_social_posts p
LEFT JOIN latest l ON l.post_id = p.id
GROUP BY p.post_type
ORDER BY reacciones_prom DESC;
