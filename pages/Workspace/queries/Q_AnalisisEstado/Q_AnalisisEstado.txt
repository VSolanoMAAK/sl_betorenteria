WITH params AS (
    SELECT
        {{ CW_Analisis.model.selectedQueryId }}::uuid
            AS query_id,

        GREATEST(
            1,
            LEAST(
                {{ CW_Analisis.model.periodDays }}::integer,
                365
            )
        ) AS period_days,

        lower(
            COALESCE(
                NULLIF(
                    {{ CW_Analisis.model.platform }}::text,
                    ''
                ),
                'all'
            )
        ) AS platform,

        lower(
            {{ appsmith.store.userEmail }}::text
        ) AS user_email
),

query_context AS (
    SELECT
        q.tenant_id,
        q.id AS query_id,
        q.name AS query_name,
        p.period_days,
        p.platform

    FROM params p

    JOIN public.sl_users u
      ON lower(u.email) = p.user_email
     AND u.is_active = TRUE

    JOIN public.sl_queries q
      ON q.tenant_id = u.tenant_id
     AND q.id = p.query_id

    LIMIT 1
),

posts_base AS (
    SELECT
        qc.tenant_id,
        qc.query_id,
        qc.query_name,
        qc.period_days,
        qc.platform AS selected_platform,

        sp.id AS post_id,
        lower(mt.platform) AS platform,

        COALESCE(
            sp.published_at,
            sp.scraped_at
        ) AS event_at,

        lm.id AS metric_id,
        lm.likes_count,
        lm.comments_count,
        lm.shares_count,
        lm.views_count,
        lm.reposts_count,
        lm.quotes_count,
        lm.fetched_at,

        CASE
            WHEN lm.id IS NULL
                THEN NULL
            ELSE
                COALESCE(lm.likes_count, 0)
                + COALESCE(lm.comments_count, 0)
                + COALESCE(lm.shares_count, 0)
                + COALESCE(lm.reposts_count, 0)
                + COALESCE(lm.quotes_count, 0)
        END AS observed_interactions

    FROM query_context qc

    JOIN public.sl_query_social_posts qsp
      ON qsp.tenant_id = qc.tenant_id
     AND qsp.query_id = qc.query_id
     AND qsp.is_active = TRUE

    JOIN public.sl_social_posts sp
      ON sp.tenant_id = qsp.tenant_id
     AND sp.id = qsp.post_id

    JOIN public.sl_monitored_targets mt
      ON mt.tenant_id = sp.tenant_id
     AND mt.id = sp.target_id

    LEFT JOIN LATERAL (
        SELECT
            m.id,
            m.likes_count,
            m.comments_count,
            m.shares_count,
            m.views_count,
            m.reposts_count,
            m.quotes_count,
            m.fetched_at
        FROM public.sl_post_metrics m
        WHERE m.tenant_id = sp.tenant_id
          AND m.post_id = sp.id
        ORDER BY
            m.fetched_at DESC NULLS LAST,
            m.id DESC
        LIMIT 1
    ) lm
      ON TRUE

    WHERE
        (
            qc.platform = 'all'
            OR lower(mt.platform) = qc.platform
        )

        AND COALESCE(
            sp.published_at,
            sp.scraped_at
        ) >=
            CURRENT_TIMESTAMP
            - (
                qc.period_days * 2
              ) * INTERVAL '1 day'

        AND COALESCE(
            sp.published_at,
            sp.scraped_at
        ) <= CURRENT_TIMESTAMP
),

classified AS (
    SELECT
        pb.*,

        CASE
            WHEN pb.event_at >=
                CURRENT_TIMESTAMP
                - pb.period_days * INTERVAL '1 day'
                THEN 'current'

            ELSE 'previous'
        END AS period_bucket

    FROM posts_base pb
),

totals AS (
    SELECT
        query_id,
        query_name,
        period_days,
        selected_platform,

        COUNT(*) FILTER (
            WHERE period_bucket = 'current'
        )::bigint AS current_posts,

        COUNT(*) FILTER (
            WHERE period_bucket = 'previous'
        )::bigint AS previous_posts,

        COUNT(metric_id) FILTER (
            WHERE period_bucket = 'current'
        )::bigint AS current_posts_with_metrics,

        COUNT(metric_id) FILTER (
            WHERE period_bucket = 'previous'
        )::bigint AS previous_posts_with_metrics,

        SUM(observed_interactions) FILTER (
            WHERE period_bucket = 'current'
        )::bigint AS current_interactions,

        SUM(observed_interactions) FILTER (
            WHERE period_bucket = 'previous'
        )::bigint AS previous_interactions,

        SUM(comments_count) FILTER (
            WHERE period_bucket = 'current'
        )::bigint AS current_comments,

        SUM(comments_count) FILTER (
            WHERE period_bucket = 'previous'
        )::bigint AS previous_comments,

        SUM(views_count) FILTER (
            WHERE period_bucket = 'current'
        )::bigint AS current_views,

        SUM(views_count) FILTER (
            WHERE period_bucket = 'previous'
        )::bigint AS previous_views,

        COUNT(DISTINCT platform) FILTER (
            WHERE period_bucket = 'current'
        )::integer AS current_platforms,

        MAX(fetched_at) FILTER (
            WHERE period_bucket = 'current'
        ) AS latest_metric_at

    FROM classified

    GROUP BY
        query_id,
        query_name,
        period_days,
        selected_platform
)

SELECT
    query_id,
    query_name,
    period_days,
    selected_platform,

    current_posts,
    previous_posts,

    current_interactions,
    previous_interactions,

    current_comments,
    previous_comments,

    current_views,
    previous_views,

    current_posts_with_metrics,
    previous_posts_with_metrics,

    CASE
        WHEN current_posts > 0
            THEN ROUND(
                (
                    current_posts_with_metrics::numeric
                    / current_posts::numeric
                ) * 100,
                2
            )
        ELSE NULL
    END AS current_metrics_coverage_pct,

    CASE
        WHEN previous_posts > 0
            THEN ROUND(
                (
                    previous_posts_with_metrics::numeric
                    / previous_posts::numeric
                ) * 100,
                2
            )
        ELSE NULL
    END AS previous_metrics_coverage_pct,

    ROUND(
        current_posts::numeric
        / period_days::numeric,
        2
    ) AS current_posts_per_day,

    ROUND(
        previous_posts::numeric
        / period_days::numeric,
        2
    ) AS previous_posts_per_day,

    CASE
        WHEN current_posts > 0
             AND current_interactions IS NOT NULL
            THEN ROUND(
                current_interactions::numeric
                / current_posts::numeric,
                2
            )
        ELSE NULL
    END AS current_interactions_per_post,

    CASE
        WHEN previous_posts > 0
             AND previous_interactions IS NOT NULL
            THEN ROUND(
                previous_interactions::numeric
                / previous_posts::numeric,
                2
            )
        ELSE NULL
    END AS previous_interactions_per_post,

    CASE
        WHEN previous_posts > 0
            THEN ROUND(
                (
                    (
                        current_posts::numeric
                        - previous_posts::numeric
                    )
                    / previous_posts::numeric
                ) * 100,
                2
            )
        ELSE NULL
    END AS posts_change_pct,

    CASE
        WHEN previous_interactions > 0
            THEN ROUND(
                (
                    (
                        current_interactions::numeric
                        - previous_interactions::numeric
                    )
                    / previous_interactions::numeric
                ) * 100,
                2
            )
        ELSE NULL
    END AS interactions_change_pct,

    CASE
        WHEN previous_comments > 0
            THEN ROUND(
                (
                    (
                        current_comments::numeric
                        - previous_comments::numeric
                    )
                    / previous_comments::numeric
                ) * 100,
                2
            )
        ELSE NULL
    END AS comments_change_pct,

    CASE
        WHEN previous_views > 0
            THEN ROUND(
                (
                    (
                        current_views::numeric
                        - previous_views::numeric
                    )
                    / previous_views::numeric
                ) * 100,
                2
            )
        ELSE NULL
    END AS views_change_pct,

    current_platforms,
    latest_metric_at,

    CURRENT_TIMESTAMP
        - period_days * INTERVAL '1 day'
        AS current_period_from,

    CURRENT_TIMESTAMP
        AS current_period_to,

    CURRENT_TIMESTAMP
        - period_days * 2 * INTERVAL '1 day'
        AS previous_period_from,

    CURRENT_TIMESTAMP
        - period_days * INTERVAL '1 day'
        AS previous_period_to

FROM totals;
