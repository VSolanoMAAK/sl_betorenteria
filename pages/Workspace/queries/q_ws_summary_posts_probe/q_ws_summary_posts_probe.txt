WITH params AS (
    SELECT
        NULLIF(
            {{ appsmith.store.selectedQueryId || '' }},
            ''
        )::uuid AS query_id,

        LOWER(
            NULLIF(
                {{ appsmith.store.userEmail || '' }},
                ''
            )
        ) AS user_email,

        CASE
            WHEN {{ CW_Resumen.model.period || '30d' }} = '24h'
                THEN INTERVAL '24 hours'

            WHEN {{ CW_Resumen.model.period || '30d' }} = '7d'
                THEN INTERVAL '7 days'

            WHEN {{ CW_Resumen.model.period || '30d' }} = '90d'
                THEN INTERVAL '90 days'

            ELSE INTERVAL '30 days'
        END AS lookback
),

authorized_query AS (
    SELECT
        q.id,
        q.tenant_id,
        q.name,
        q.formula_text

    FROM public.sl_queries q

    INNER JOIN public.sl_users u
        ON u.tenant_id = q.tenant_id

    CROSS JOIN params p

    WHERE q.id = p.query_id
      AND LOWER(u.email) = p.user_email
      AND u.is_active = TRUE

    LIMIT 1
),

target_map AS (
    SELECT
        qt.target_id,
        mt.platform,
        mt.handle

    FROM authorized_query q

    INNER JOIN public.sl_query_targets qt
        ON qt.tenant_id = q.tenant_id
       AND qt.query_id = q.id

    INNER JOIN public.sl_monitored_targets mt
        ON mt.tenant_id = qt.tenant_id
       AND mt.id = qt.target_id
),

candidate_posts AS (
    SELECT
        p.id AS post_id,
        p.target_id,
        tm.platform,
        tm.handle,
        p.external_post_id,
        p.content_text,
        p.post_url,
        p.post_type,

        COALESCE(
            p.published_at,
            p.scraped_at
        ) AS event_at

    FROM authorized_query q

    INNER JOIN target_map tm
        ON TRUE

    INNER JOIN public.sl_social_posts p
        ON p.tenant_id = q.tenant_id
       AND p.target_id = tm.target_id

    CROSS JOIN params prm

    WHERE COALESCE(
        p.published_at,
        p.scraped_at
    ) >= CURRENT_TIMESTAMP - prm.lookback
),

query_terms AS (
    SELECT
        g.group_no,
        t.operator,
        t.term,
        t.position

    FROM authorized_query q

    INNER JOIN public.sl_query_groups g
        ON g.tenant_id = q.tenant_id
       AND g.query_id = q.id

    INNER JOIN public.sl_query_terms t
        ON t.tenant_id = g.tenant_id
       AND t.group_id = g.id
)

SELECT
    q.id AS query_id,
    q.name AS query_name,
    q.formula_text,

    (
        SELECT COUNT(*)::integer
        FROM target_map
    ) AS configured_targets,

    (
        SELECT COUNT(*)::integer
        FROM candidate_posts
    ) AS ingested_posts_in_period,

    (
        SELECT COUNT(
            DISTINCT platform
        )::integer
        FROM candidate_posts
    ) AS active_platforms,

    (
        SELECT MAX(event_at)
        FROM candidate_posts
    ) AS last_post_activity,

    COALESCE(
        (
            SELECT jsonb_agg(
                jsonb_build_object(
                    'group_no',
                    qt.group_no,

                    'operator',
                    qt.operator,

                    'term',
                    qt.term,

                    'position',
                    qt.position
                )
                ORDER BY
                    qt.group_no,
                    qt.position
            )

            FROM query_terms qt
        ),
        '[]'::jsonb
    ) AS query_terms,

    COALESCE(
        (
            SELECT jsonb_agg(
                jsonb_build_object(
                    'post_id',
                    cp.post_id,

                    'platform',
                    cp.platform,

                    'handle',
                    cp.handle,

                    'content_text',
                    cp.content_text,

                    'post_url',
                    cp.post_url,

                    'post_type',
                    cp.post_type,

                    'event_at',
                    cp.event_at
                )
                ORDER BY cp.event_at DESC
            )

            FROM (
                SELECT *
                FROM candidate_posts

                ORDER BY event_at DESC

                LIMIT 20
            ) cp
        ),
        '[]'::jsonb
    ) AS sample_posts

FROM authorized_query q;
