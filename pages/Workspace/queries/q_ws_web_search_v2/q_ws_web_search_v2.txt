{{
  (() => {
    const parts =
      new Intl.DateTimeFormat(
        'en-US',
        {
          timeZone: 'America/Mexico_City',
          year: 'numeric',
          month: '2-digit',
          day: '2-digit'
        }
      ).formatToParts(new Date());

    const values = {};

    parts.forEach(function (part) {
      values[part.type] = part.value;
    });

    const today =
      values.year + '-' +
      values.month + '-' +
      values.day;

    return {
      queryId:
        appsmith.store.selectedQueryId ||
        appsmith.store.activeQueryId ||
        "",

      userEmail:
        appsmith.store.userEmail ||
        "",

      dateFrom:
        today,

      dateTo:
        today,

      originFilter:
        ""
    };
  })()
}}
