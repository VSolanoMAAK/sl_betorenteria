WITH latest_metrics AS (
    SELECT DISTINCT ON (pm.post_id)
        pm.post_id,
        pm.tenant_id,
        COALESCE(pm.likes_count, 0) AS likes,
        COALESCE(pm.comments_count, 0) AS comments,
        COALESCE(pm.shares_count, 0) AS shares,
        COALESCE(pm.views_count, 0) AS views,
        COALESCE(pm.reposts_count, 0) AS reposts,
        COALESCE(pm.quotes_count, 0) AS quotes,
        pm.fetched_at
    FROM public.sl_post_metrics pm
    WHERE pm.tenant_id =
        'e494efe2-07f5-4197-ac77-5f2f0bc59a92'::uuid
    ORDER BY
        pm.post_id,
        pm.fetched_at DESC
)

SELECT
    sp.id,
    t.platform,
    t.handle,
    sp.content_text,
    sp.published_at,
    sp.post_url,
    sp.post_type,

    lm.likes,
    lm.comments,
    lm.shares,
    lm.views,
    lm.reposts,
    lm.quotes,

    (
        COALESCE(lm.likes, 0) +
        COALESCE(lm.comments, 0) * 2 +
        COALESCE(lm.shares, 0) * 3 +
        COALESCE(lm.reposts, 0) * 3 +
        COALESCE(lm.quotes, 0) * 3
    ) AS impact_score

FROM public.sl_social_posts sp

JOIN public.sl_monitored_targets t
  ON t.id = sp.target_id
 AND t.tenant_id = sp.tenant_id

LEFT JOIN latest_metrics lm
  ON lm.post_id = sp.id
 AND lm.tenant_id = sp.tenant_id

WHERE sp.tenant_id =
    'e494efe2-07f5-4197-ac77-5f2f0bc59a92'::uuid

  AND sp.published_at >= CURRENT_DATE - INTERVAL '30 days'

ORDER BY
    impact_score DESC,
    sp.published_at DESC

LIMIT 20;
