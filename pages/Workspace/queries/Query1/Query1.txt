function formatDrawerMetric(value) {
    if (
        value === null ||
        value === undefined ||
        value === ''
    ) {
        return '—';
    }

    var number = Number(value);

    if (!isFinite(number)) {
        return '—';
    }

    try {
        return new Intl.NumberFormat('es-MX').format(number);
    } catch (error) {
        return String(number);
    }
}
