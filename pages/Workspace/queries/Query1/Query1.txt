WITH params AS (
    SELECT
        u.tenant_id,

        NULLIF(
            {{
                CW_Constructor.model.selectedQueryId
                || appsmith.store.selectedQueryId
                || appsmith.store.activeQueryId
                || ""
            }},
            ''
        )::uuid AS query_id,

        NULLIF(
            {{
                CW_Constructor.model.selectedQueryId
                || ""
            }},
            ''
        )::text AS cw_selected_query_id,

        NULLIF(
            {{
                appsmith.store.selectedQueryId
                || ""
            }},
            ''
        )::text AS store_selected_query_id,

        NULLIF(
            {{
                appsmith.store.activeQueryId
                || ""
            }},
            ''
        )::text AS store_active_query_id

    FROM public.sl_users u

    WHERE lower(u.email) =
          lower({{ appsmith.store.userEmail || "" }})

      AND u.is_active = TRUE

    LIMIT 1
),

query_posts AS (
    SELECT
        qsp.tenant_id,
        qsp.query_id,
        qsp.post_id,
        qsp.is_active

    FROM params prm

    JOIN public.sl_query_social_posts qsp
      ON qsp.tenant_id = prm.tenant_id
     AND qsp.query_id = prm.query_id
),

active_query_posts AS (
    SELECT
        qp.*

    FROM query_posts qp

    WHERE qp.is_active = TRUE
),

query_comments AS (
    SELECT
        c.id,
        c.post_id,
        c.author_id,
        c.author_name,
        c.author_url,
        c.content_text,
        c.published_at

    FROM params prm

    JOIN active_query_posts qp
      ON qp.tenant_id = prm.tenant_id
     AND qp.query_id = prm.query_id

    JOIN public.sl_comments c
      ON c.tenant_id = qp.tenant_id
     AND c.post_id = qp.post_id
),

query_web_keywords AS (
    SELECT
        qwk.tenant_id,
        qwk.query_id,
        qwk.keyword_id

    FROM params prm

    JOIN public.sl_query_web_keywords qwk
      ON qwk.tenant_id = prm.tenant_id
     AND qwk.query_id = prm.query_id
),

query_web_mentions AS (
    SELECT
        wm.id,
        wm.keyword_id,
        wm.source_name,
        wm.source_domain,
        wm.source_url

    FROM params prm

    JOIN query_web_keywords qwk
      ON qwk.tenant_id = prm.tenant_id
     AND qwk.query_id = prm.query_id

    JOIN public.sl_web_mentions wm
      ON wm.tenant_id = qwk.tenant_id
     AND wm.keyword_id = qwk.keyword_id
)

SELECT
    prm.tenant_id,

    prm.query_id,

    prm.cw_selected_query_id,

    prm.store_selected_query_id,

    prm.store_active_query_id,

    (
        SELECT COUNT(*)
        FROM query_posts
    ) AS query_post_relations,

    (
        SELECT COUNT(*)
        FROM active_query_posts
    ) AS active_post_relations,

    (
        SELECT COUNT(*)
        FROM query_comments
    ) AS comments_found,

    (
        SELECT COUNT(*)
        FROM query_comments
        WHERE NULLIF(
            btrim(author_name),
            ''
        ) IS NOT NULL
    ) AS comments_with_author_name,

    (
        SELECT COUNT(*)
        FROM query_web_keywords
    ) AS web_keywords_found,

    (
        SELECT COUNT(*)
        FROM query_web_mentions
    ) AS web_mentions_found

FROM params prm;
