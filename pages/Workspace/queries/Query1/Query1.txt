WITH definitions AS (
    SELECT
        q.tenant_id,
        q.id AS query_id,

        jsonb_build_object(
            'query_id',
            q.id,

            'name',
            q.name,

            'formula_text',
            q.formula_text,

            'use_web',
            q.use_web,

            'settings',
            q.settings,

            'groups',
            COALESCE(
                (
                    SELECT jsonb_agg(
                        jsonb_build_object(
                            'group_no',
                            g.group_no,

                            'join_operator',
                            g.join_operator,

                            'include',
                            COALESCE(
                                (
                                    SELECT jsonb_agg(
                                        t.term
                                        ORDER BY t.position
                                    )
                                    FROM public.sl_query_terms t
                                    WHERE t.tenant_id = g.tenant_id
                                      AND t.group_id = g.id
                                      AND t.operator = 'include'
                                ),
                                '[]'::jsonb
                            ),

                            'and',
                            COALESCE(
                                (
                                    SELECT jsonb_agg(
                                        t.term
                                        ORDER BY t.position
                                    )
                                    FROM public.sl_query_terms t
                                    WHERE t.tenant_id = g.tenant_id
                                      AND t.group_id = g.id
                                      AND t.operator = 'and'
                                ),
                                '[]'::jsonb
                            ),

                            'and_not',
                            COALESCE(
                                (
                                    SELECT jsonb_agg(
                                        t.term
                                        ORDER BY t.position
                                    )
                                    FROM public.sl_query_terms t
                                    WHERE t.tenant_id = g.tenant_id
                                      AND t.group_id = g.id
                                      AND t.operator = 'and_not'
                                ),
                                '[]'::jsonb
                            )
                        )
                        ORDER BY g.group_no
                    )
                    FROM public.sl_query_groups g
                    WHERE g.tenant_id = q.tenant_id
                      AND g.query_id = q.id
                ),
                '[]'::jsonb
            ),

            'targets',
            COALESCE(
                (
                    SELECT jsonb_agg(
                        jsonb_build_object(
                            'target_id',
                            t.id,

                            'platform',
                            t.platform,

                            'handle',
                            t.handle,

                            'external_profile_id',
                            t.external_profile_id
                        )
                        ORDER BY
                            t.platform,
                            t.handle,
                            t.id
                    )
                    FROM public.sl_query_targets qt

                    JOIN public.sl_monitored_targets t
                      ON t.tenant_id = qt.tenant_id
                     AND t.id = qt.target_id

                    JOIN public.sl_platforms p
                      ON p.code = t.platform
                     AND p.is_active = TRUE

                    WHERE qt.tenant_id = q.tenant_id
                      AND qt.query_id = q.id
                      AND t.status = 'active'
                ),
                '[]'::jsonb
            ),

            'web_keywords',
            COALESCE(
                (
                    SELECT jsonb_agg(
                        jsonb_build_object(
                            'keyword_id',
                            wk.id,

                            'phrase',
                            wk.phrase
                        )
                        ORDER BY
                            lower(wk.phrase),
                            wk.id
                    )
                    FROM public.sl_query_web_keywords qwk

                    JOIN public.sl_web_keywords wk
                      ON wk.tenant_id = qwk.tenant_id
                     AND wk.id = qwk.keyword_id

                    WHERE qwk.tenant_id = q.tenant_id
                      AND qwk.query_id = q.id
                ),
                '[]'::jsonb
            )
        ) AS query_definition

    FROM public.sl_queries q

    WHERE q.status = 'active'
      AND q.tenant_id =
          'e494efe2-07f5-4197-ac77-5f2f0bc59a92'::uuid
      AND q.id =
          '0542957e-d74f-423f-9c8a-ca569d48dda3'::uuid

      AND (
          EXISTS (
              SELECT 1
              FROM public.sl_query_targets qt

              JOIN public.sl_monitored_targets t
                ON t.tenant_id = qt.tenant_id
               AND t.id = qt.target_id

              JOIN public.sl_platforms p
                ON p.code = t.platform
               AND p.is_active = TRUE

              WHERE qt.tenant_id = q.tenant_id
                AND qt.query_id = q.id
                AND t.status = 'active'
          )

          OR

          (
              q.use_web = TRUE
              AND EXISTS (
                  SELECT 1
                  FROM public.sl_query_web_keywords qwk
                  WHERE qwk.tenant_id = q.tenant_id
                    AND qwk.query_id = q.id
              )
          )
      )
),

current_definition AS (
    SELECT
        tenant_id,
        query_id,
        query_definition,
        md5(query_definition::text) AS current_definition_hash
    FROM definitions
),

last_success AS (
    SELECT
        r.definition_hash AS last_success_definition_hash,
        r.started_at,
        r.finished_at
    FROM public.sl_query_runs r
    WHERE r.tenant_id =
        'e494efe2-07f5-4197-ac77-5f2f0bc59a92'::uuid
      AND r.query_id =
        '0542957e-d74f-423f-9c8a-ca569d48dda3'::uuid
      AND r.status = 'succeeded'
    ORDER BY
        r.finished_at DESC NULLS LAST,
        r.started_at DESC,
        r.id DESC
    LIMIT 1
)

SELECT
    cd.query_id,
    cd.current_definition_hash,
    ls.last_success_definition_hash,
    cd.current_definition_hash =
        ls.last_success_definition_hash
        AS hashes_match,
    CASE
        WHEN ls.last_success_definition_hash IS NULL
            THEN 'first_run'
        WHEN cd.current_definition_hash
             IS DISTINCT FROM
             ls.last_success_definition_hash
            THEN 'definition_changed'
        ELSE 'incremental'
    END AS expected_execution_mode
FROM current_definition cd
LEFT JOIN last_success ls
    ON TRUE;
