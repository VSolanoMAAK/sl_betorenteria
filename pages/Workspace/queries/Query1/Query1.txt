SELECT
    COUNT(*) AS grupos_duplicados,
    COALESCE(SUM(cantidad - 1), 0) AS filas_duplicadas
FROM (
    SELECT
        tenant_id,
        actor_id,
        external_post_id,
        COUNT(*) AS cantidad
    FROM public.sl_actor_posts
    WHERE tenant_id = 'e494efe2-07f5-4197-ac77-5f2f0bc59a92'::uuid
    GROUP BY tenant_id, actor_id, external_post_id
    HAVING COUNT(*) > 1
) d;
