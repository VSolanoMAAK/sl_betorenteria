WITH matched AS (
    SELECT
        qsp.post_id,
        mt.platform
    FROM public.sl_query_social_posts qsp
    JOIN public.sl_social_posts sp
      ON sp.tenant_id = qsp.tenant_id
     AND sp.id = qsp.post_id
    JOIN public.sl_monitored_targets mt
      ON mt.tenant_id = sp.tenant_id
     AND mt.id = sp.target_id
    WHERE qsp.tenant_id =
          'e494efe2-07f5-4197-ac77-5f2f0bc59a92'::uuid
      AND qsp.query_id =
          '0542957e-d74f-423f-9c8a-ca569d48dda3'::uuid
      AND qsp.is_active = TRUE
),
summary AS (
    SELECT
        platform,
        COUNT(DISTINCT post_id) AS matched_posts,
        1 AS sort_order
    FROM matched
    GROUP BY platform

    UNION ALL

    SELECT
        'TOTAL' AS platform,
        COUNT(DISTINCT post_id) AS matched_posts,
        2 AS sort_order
    FROM matched
)

SELECT
    platform,
    matched_posts
FROM summary
ORDER BY
    sort_order,
    platform;
